cmake_minimum_required(VERSION 3.25)
project(rttelemetry C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Warnings shared by every build; src/core adds stricter ones itself.
add_compile_options(-Wall -Wextra -Werror -Wshadow -Wundef -Wdouble-promotion)

if(CMAKE_CROSSCOMPILING)
  add_subdirectory(src/core)
  add_subdirectory(src)                # firmware image
else()
  set(SANITIZE "" CACHE STRING "Comma-separated sanitizers for host builds (address,undefined | thread)")
  option(COVERAGE "Instrument host build for coverage" OFF)
  option(FUZZ "Build libFuzzer targets (requires LLVM clang)" OFF)
  if(SANITIZE)
    add_compile_options(-fsanitize=${SANITIZE} -fno-omit-frame-pointer -fno-sanitize-recover=all)
    add_link_options(-fsanitize=${SANITIZE})
  endif()
  if(COVERAGE)
    add_compile_options(--coverage -O0)
    add_link_options(--coverage)
  endif()
  enable_testing()
  add_subdirectory(src/core)
  add_subdirectory(tests)
endif()
